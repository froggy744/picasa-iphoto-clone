cmake_minimum_required(VERSION 3.16)
project(pic_nfs_helper C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBNFS REQUIRED IMPORTED_TARGET libnfs)
add_executable(pic-nfs-helper pic-nfs-helper-portable.c)
target_link_libraries(pic-nfs-helper PRIVATE PkgConfig::LIBNFS)
if(MSVC)
  target_compile_options(pic-nfs-helper PRIVATE /W4)
else()
  target_compile_options(pic-nfs-helper PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(APPLE)
  set_target_properties(pic-nfs-helper PROPERTIES BUILD_RPATH "@executable_path")
elseif(UNIX)
  set_target_properties(pic-nfs-helper PROPERTIES BUILD_RPATH "$ORIGIN")
endif()
install(TARGETS pic-nfs-helper RUNTIME DESTINATION libexec)
